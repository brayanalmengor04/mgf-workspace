# Variables de correo Gmail para Railway (servicio mgf-workspace)
# Misma idea que tu portfolio (Nodemailer service: "gmail" → puerto 465 SSL).
# https://github.com/brayanalmengor04/brayanalmengor04.github.io

MAIL_MAILER=smtp
MAIL_HOST=smtp.gmail.com
MAIL_PORT=465
MAIL_SCHEME=smtps
MAIL_TIMEOUT=15

# Equivalente a EMAIL_USER / EMAIL_PASS del portfolio
MAIL_USERNAME=tu_correo@gmail.com
MAIL_PASSWORD=contraseña_aplicacion_16_caracteres

MAIL_FROM_ADDRESS=tu_correo@gmail.com
MAIL_FROM_NAME=MGF Workspace

QUEUE_CONNECTION=sync

# --- Notas ---

# 1. Gmail: contraseña de aplicación (no la contraseña normal)
#    https://myaccount.google.com/apppasswords
#
# 2. Si tenías MAIL_PORT=587 o MAIL_ENCRYPTION=tls, cámbialos a 465/smtps.
#    La app también fuerza 465 para hosts gmail.com automáticamente.
#
# 3. Sin comillas en Railway (no "smtp", no "false").
#
# 4. Elimina SESSION_DOMAIN=null si existe.
#
# 5. Diagnóstico local con vars de prod: just prod-mail-diagnose
