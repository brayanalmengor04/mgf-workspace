# Variables de correo para Railway (servicio mgf-workspace)
# Copia cada clave en Railway → Variables. NO subas valores reales al repositorio.

# =============================================================================
# OPCIÓN A — Resend (RECOMENDADO en Railway / producción)
# Gmail SMTP suele dar TIMEOUT desde servidores cloud (puerto 587 bloqueado).
# Resend usa HTTPS y funciona en Railway.
# =============================================================================
#
# 1. Crea cuenta en https://resend.com
# 2. API Keys → Create API Key
# 3. En Railway:
#
# MAIL_MAILER=resend
# RESEND_API_KEY=re_xxxxxxxx
# MAIL_FROM_ADDRESS=onboarding@resend.dev
#   (pruebas; luego verifica tu dominio en Resend y usa tu@tudominio.com)
# MAIL_FROM_NAME=MGF Workspace
#
# Diagnóstico: just prod-mail-diagnose
# Logs al arrancar: busca "mail-probe:" en Deploy Logs

# =============================================================================
# OPCIÓN B — Gmail SMTP (funciona en local; en Railway a menudo falla)
# =============================================================================

MAIL_MAILER=smtp
# Valor: smtp (local) o resend (Railway)

MAIL_SCHEME=smtp
# Valor: smtp (puerto 587). Alternativa en Railway: smtps + puerto 465

MAIL_HOST=smtp.gmail.com
MAIL_PORT=587
MAIL_TIMEOUT=15

MAIL_USERNAME=
# Correo Gmail completo

MAIL_PASSWORD=
# Contraseña de aplicación Google (16 caracteres)
# https://myaccount.google.com/apppasswords

MAIL_FROM_ADDRESS=
MAIL_FROM_NAME=MGF Workspace

# Si Gmail falla en Railway, prueba antes de Resend:
# MAIL_SCHEME=smtps
# MAIL_PORT=465

QUEUE_CONNECTION=sync
# En producción sin worker de cola, usa sync para envío inmediato

# --- Notas ---

# 1. Sin comillas en los valores de Railway (no uses "false" ni "smtp" entre comillas).
# 2. Agrega MAIL_SCHEME=smtp (MAIL_ENCRYPTION=tls solo es respaldo).
# 3. Elimina SESSION_DOMAIN=null si la tienes.
# 4. Tras cambiar variables: Redeploy.
# 5. El entrypoint regenera .env en cada arranque con MAIL_* y RESEND_*.
